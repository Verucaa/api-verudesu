KENAPA VERUdesu

Didesain untuk developer yang ingin kreatif, bukan sibuk konfigurasi.

CARA PAKAI

Panggil API dari aplikasi apa pun — cukup kirim request ke endpoint, lalu olah respons JSON-nya.

1

LIHAT CATALOG

GET /api/endpoints — dapatkan daftar semua endpoint, method, dan parameter yang tersedia.

2

KIRIM REQUEST

Panggil /<kategori>/<endpoint> dengan method GET atau POST sesuai schema params.

3

OLAH RESPONS

Respons standar { status, creator, result } — tinggal di-parse di kode aplikasimu.

// CONTOH PEMANGGILAN API — pilih bahasamu
GET /anime/otakudesu/search?query=…
# Ambil daftar semua endpoint
curl https://your-api.com/api/endpoints

# Panggil endpoint dengan parameter
curl -G "https://your-api.com/anime/otakudesu/search" \
  --data-urlencode "query=naruto"
const res = await fetch("https://your-api.com/anime/otakudesu/search?query=" +
  encodeURIComponent("naruto"));
const data = await res.json();

if (data.status) console.log(data.result);
else console.error(data.message);
import requests

r = requests.get("https://your-api.com/anime/otakudesu/search",
                 params={"query": "naruto"})
data = r.json()

print(data["status"], data["result"])
<?php
$ch = curl_init("https://your-api.com/anime/otakudesu/search");
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);

  "query" => "naruto",

package main

import (
  "fmt"
  "io"
  "net/http"
  "net/url"
)

func main() {
  u := "https://your-api.com/anime/otakudesu/search?query=" +
    url.QueryEscape("naruto")
  resp, _ := http.Get(u)
  defer resp.Body.Close()
  body, _ := io.ReadAll(resp.Body)
  fmt.Println(string(body))
}
fetch("https://your-api.com/anime/otakudesu/search?query=naruto")
  .then((r) => r.json())
  .then((data) => {
    if (data.status) console.log(data.result);
  });
